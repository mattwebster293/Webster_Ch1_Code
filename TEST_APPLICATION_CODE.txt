df_windows <- aoi_df %>%
  arrange(watershed, year, TP_ID) %>%												#get everything in sequential order
  group_by(watershed, year) %>%														#create groups (watershed-year)
  group_modify(~ {																	#operate per watershed-year
  
    n <- nrow(.x)																	#".x" is "watershed-year[i]"
	
    window_id <- seq_len(n - 25 + 1)												#define windows (24 less than max TP_ID)

    tibble(
      window_id   = window_id,
      TP_ID_start = window_id,														#start here
      TP_ID_end   = window_id + 24,													#progress upstream +24 TP_IDs (25 TP_ID total = ~250m)

      TO_mean     = slide_dbl(.x$TO,   mean, .before = 0, .after = 24)[window_id],
      TO_sd       = slide_dbl(.x$TO,   sd,   .before = 0, .after = 24)[window_id],

      RIPA_mean   = slide_dbl(.x$RIPA, mean, .before = 0, .after = 24)[window_id],
      RIPA_sd     = slide_dbl(.x$RIPA, sd,   .before = 0, .after = 24)[window_id],

    )
  }) %>%
  ungroup()

# SURFACE WATER AREA

predict_SW_window <- function(TO_mean, RIPA_mean, UDA_mean,
                              damDensity,
                              stream, year,
                              coefs_SW_fixed,
                              coefs_SW_random_stream,
                              coefs_SW_random_year,
                              link = "log") {
  
  eta <- coefs_SW_fixed["alpha"] +
    coefs_SW_fixed["beta_incis"]     * TO_mean +
    coefs_SW_fixed["beta_RIPA"]      * RIPA_mean +
    coefs_SW_fixed["beta_UDA"]       * UDA_mean +
    coefs_SW_fixed["beta_damDen"]    * damDensity +
    coefs_SW_random_stream[as.character(stream)] +
    coefs_SW_random_year[as.character(year)]
	
  SW <- exp(eta)
  
  SW
}
